1. Project Context & Environment Architecture

The project is an automotive-optimized music player built using React Native with a robust native Android Java/Kotlin engineering layer. The app must execute heavy audio asset handling and platform tracking reliably on mid-tier hardware (e.g., SM-A107F running Android 11, API Level 30).

Current Core Stack

  • React Native: Modern Architecture (Bridgeless / TurboModules enabled by default).
  • Build Tool: Gradle 9.4.1.
  • Execution SDK: Compile/Target Android SDK 34.0.0.
  • Native Layer Runtime Requirements: Extreme startup stability requiring strict explicit synchronization hooks for dependencies like SoLoader before React Native assets load.
Critical OS Constraint solved: STANDALONE APK Lifecycles run faster than continuous debug wrappers. C++ JNI layout bridges require SoLoader.init() execution immediately inside the classloading routine via attachBaseContext(base: Context?) within the root Application wrapper class before static feature configuration mapping runs, avoiding unhandled ExceptionInInitializerError loops.

B. Foreground Playback Lifecycle Execution

File location path: android/app/src/main/java/com/automusicplayer/media/PlaybackService.kt

package com.automusicplayer.media

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Intent
import android.os.Build
import android.os.IBinder
import androidx.core.app.NotificationCompat

class PlaybackService : Service() {

    private val CHANNEL_ID = "amp_playback_channel"

    override fun onCreate() {
        // Strict Enforcement: Initialize foreground assets at the absolute start of onCreate()
        createNotificationChannel()
        val notification = createNotification("Initializing Playback Engine...")
        startForeground(1, notification)
        
        super.onCreate()
        // Proceed with MediaLibrarySession binding...
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID, "Audio Playback Service",
                NotificationManager.IMPORTANCE_LOW
            )
            val manager = getSystemService(NotificationManager::class.java)
            manager?.createNotificationChannel(channel)
        }
    }

    private fun createNotification(contentText: String): Notification {
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("Auto Music Player")
            .setContentText(contentText)
            .setSmallIcon(android.R.drawable.ic_media_play)
            .build()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null
}
